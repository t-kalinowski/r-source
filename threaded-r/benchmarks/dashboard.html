Threaded scaling

thr <- subset(results, mode == "mtlapply")
if (nrow(thr)) {
  base <- subset(results, mode == "lapply", c("case", "elapsed_s"))
  names(base)[2] <- "serial_elapsed_s"
  sp <- merge(thr, base, by = "case", all.x = TRUE)
  sp$speedup <- sp$serial_elapsed_s / sp$elapsed_s
  sp$efficiency <- sp$speedup / sp$threads
  print(sp[, c("case", "threads", "elapsed_s", "serial_elapsed_s", "speedup", "efficiency")])

  oldpar <- par(no.readonly = TRUE)
  on.exit(par(oldpar), add = TRUE)
  for (case in unique(sp$case)) {
    sub <- sp[sp$case == case, ]
    plot(sub$threads, sub$speedup, type = "b", main = paste("Speedup:", case),
         xlab = "threads", ylab = "speedup")
  }
} else {
  cat("No threaded rows found.\n")
}
